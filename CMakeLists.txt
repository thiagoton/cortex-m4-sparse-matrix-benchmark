# CROSS COMPILER SETTING
SET(CMAKE_SYSTEM_NAME Generic)
CMAKE_MINIMUM_REQUIRED (VERSION 3.10.0)

SET(CMAKE_TOOLCHAIN_FILE ${CMAKE_CURRENT_SOURCE_DIR}/cmake/armgcc.cmake)

project(hello_world)

# THE VERSION NUMBER
SET (Tutorial_VERSION_MAJOR 1)
SET (Tutorial_VERSION_MINOR 0)

# ENABLE ASM
ENABLE_LANGUAGE(ASM)

SET(CMAKE_STATIC_LIBRARY_PREFIX)
SET(CMAKE_STATIC_LIBRARY_SUFFIX)

SET(CMAKE_EXECUTABLE_LIBRARY_PREFIX)
SET(CMAKE_EXECUTABLE_LIBRARY_SUFFIX)

SET(EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/_deploy/${CMAKE_BUILD_TYPE})
SET(LIBRARY_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/_deploy/${CMAKE_BUILD_TYPE})
SET(REPO_ROOT ${PROJECT_SOURCE_DIR}/)
SET(BUILD_PATH ${CMAKE_BINARY_DIR})
SET(INCLUDE_CMAKE_DIR ${REPO_ROOT}/cmake)

set(MCUX_SDK_PROJECT_NAME hello_world.elf)

include(${INCLUDE_CMAKE_DIR}/flags.cmake)

include(${INCLUDE_CMAKE_DIR}/config.cmake)

add_subdirectory(src)
